# Thorium series patches that Slipstream SKIPS, one per line, matching the
# entries in third_party/thorium/patch_scripts/series/series.
# Rationale for each is recorded in docs/DECISIONS.md.

# Hardcodes Google's shared public API key + client id/secret — ToS liability,
# and CWS install/update need no keys at all. Slipstream ships keyless.
other/thorium-default-api-keys.patch

# SECURITY-NEGATIVE: disables Windows Mark-of-the-Web / Attachment Manager
# quarantine on downloads. Unacceptable for a security-positioned browser.
other/disable-download-quarantine.patch

# SECURITY-NEGATIVE: re-enables insecure (HTTP-downgraded) downloads.
other/allow-insecure-downloads.patch

# Anti-debloat: Thorium turns the What's New page ON by default.
other/enable-whats-new-by-default.patch

# Anti-debloat: Thorium turns Chrome Labs ON by default.
other/enable-chrome-labs-by-default.patch

# SECURITY-NEGATIVE (pillar audit 2026-08-30, docs/PILLARS.md B):
# Appends arbitrary switches from a user-writable %APPDATA% conf file into
# the browser command line — a no-admin persistence/desandbox vector.
other/windows-thorium-flags-conf.patch

# SECURITY-NEGATIVE: suppresses Chromium's warning infobar for dangerous
# flags (part of the anti-tamper trio with the conf exclusion above and the
# --load-extension devmode gate patch).
other/relax-bad-flags-warning.patch

# SECURITY: resurrects the FTP client Chromium removed (unmaintained,
# plaintext protocol, expanded attack surface). Product-identity call:
# Slipstream does not ship FTP. Run apply-order dry run on each rebase.
other/ftp-support-thorium.patch
